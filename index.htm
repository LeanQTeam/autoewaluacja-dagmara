    alert('Błąd: ' + err.message);

  }

}

 

function renderLetter(p) {

  clearInterval(loadingInterval);

  document.getElementById('loadingOverlay').classList.remove('show');

 

  const pill = document.getElementById('scorePill');

  pill.className = 'score-pill ' + (p.poziom_css || 's4');

  pill.textContent = 'Ocena: ' + p.ocena;

 

  const mocne = (p.mocne || []).map(m => `<span class="tag plus">${m}</span>`).join('');

  const poprawki = (p.poprawki || []).map(m => `<span class="tag minus">${m}</span>`).join('');

  const formalClass = p.formalne_ok ? 'formal' : 'formal fail';

  const formalText = p.formalne_ok ? 'Wymogi formalne spełnione' : (p.formalne_uwagi || 'Sprawdź wymogi formalne');

 

  document.getElementById('letterContent').innerHTML = `

    <p class="letter-salutation">${p.salutation || ''}</p>

    <p class="letter-paragraph">${p.akapit_otwarcie || ''}</p>

    <div class="letter-section-label">Uzasadnienie oceny</div>

    <div class="letter-highlight">${p.uzasadnienie || ''}</div>

    <div class="letter-section-label">Mocne strony</div>

    <div class="tag-row">${mocne}</div>

    <div class="letter-section-label">Obszary do dalszego rozwoju</div>

    <div class="tag-row">${poprawki}</div>

    <div class="letter-section-label">Wymogi formalne</div>

    <div class="tag-row"><span class="tag ${formalClass}">${formalText}</span></div>

    <p class="letter-paragraph letter-closing">${p.akapit_zamkniecie || ''}</p>

  `;

  document.getElementById('letterContent').classList.add('show');

 

  letterPlainText = buildPlainText(p);

  document.getElementById('copyBtn').style.display = 'flex';

  document.getElementById('resetBtn').style.display = 'flex';

}

 

function buildPlainText(p) {

  const lines = [

    p.salutation || '', '',

    p.akapit_otwarcie || '', '',

    'UZASADNIENIE OCENY (ocena: ' + p.ocena + ')',

    p.uzasadnienie || '', '',

    'MOCNE STRONY',

    ...(p.mocne || []).map(m => '+ ' + m), '',

    'OBSZARY DO DALSZEGO ROZWOJU',

    ...(p.poprawki || []).map(m => '- ' + m), '',

    'WYMOGI FORMALNE',

    p.formalne_ok ? 'Spełnione.' : (p.formalne_uwagi || 'Sprawdź wymogi.'), '',

    p.akapit_zamkniecie || ''

  ];

  return lines.join('\n');

}

 

function copyLetter() {

  const ta = document.createElement('textarea');

  ta.value = letterPlainText;

  ta.style.cssText = 'position:fixed;top:-9999px;left:-9999px;opacity:0';

  document.body.appendChild(ta);

  ta.select();

  ta.setSelectionRange(0, 99999);

  try {

    document.execCommand('copy');

    const t = document.getElementById('toast');

    t.classList.add('show');

    setTimeout(() => t.classList.remove('show'), 2200);

  } catch(e) {

    alert('Nie udało się skopiować. Zaznacz tekst ręcznie.');

  }

  document.body.removeChild(ta);

}
